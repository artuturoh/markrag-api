import requests

response = requests.post(
    "https://markrag-api.onrender.com/v1/extract",
    headers={"X-RapidAPI-Key": "YOUR_RAPIDAPI_KEY"},
    json={
        "url": "https://example.com/blog/rag-pipeline",
        "options": {"strip_links": True}
    }
)

data = response.json()["data"]
print(f"Title: {data['title']} (Saved {data['metadata']['savingsPercentage']}%)")
print(data["markdown"])

const { data } = await res.json();
